\section{Question}
\begin{frame}{Does public broadband access affect youth depressive symptoms?}
  \SlideGroup{Population and proposed outcome}
  Rural residents aged 15--29 and the nine-item ENDES PHQ-9 score for symptoms over the previous two weeks.

  \medskip
  \SlideGroup{Candidate effect of local activation}
  Estimate the average difference in PHQ-9 points at a supported horizon after
  a designated institution gains usable service, relative to activation after the interview.
  The comparison group, event window and weights are open for coauthor review.
\end{frame}

\section{Service chronology}
\begin{frame}{Trace one facility from designation to service}
  \SlideGroup{First proposed case}
  A school in Racracalla, Junín. Trace the locality and institution across the roster, regional records, and dated facility evidence.

  \medskip
  \begin{itemize}
    \item Record designation, regional operations, institutional acceptance and observed delivery as distinct events.
    \item Track public Wi-Fi, household connectivity and individual use as separate measures.
    \item Cite each event's source and record date bounds. Check those bounds against actual interviews and the symptom recall period.
  \end{itemize}
\end{frame}
